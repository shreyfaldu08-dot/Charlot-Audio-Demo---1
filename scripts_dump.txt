        (function () {
            'use strict';
            const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
            
            const section = document.getElementById('trust');
            const cards = document.querySelectorAll('.trust-card');
            if (!section) return;

            const observer = new IntersectionObserver((entries) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        section.classList.add('revealed');
                        if (!prefersReducedMotion) {
                            cards.forEach((c, i) => {
                                setTimeout(() => c.classList.add('revealed'), 100 + (i * 150));
                            });
                        } else {
                            cards.forEach(c => c.classList.add('revealed'));
                        }
                    }
                });
            }, { rootMargin: '0px 0px -15% 0px' });
            observer.observe(section);

            const themeObserver = new IntersectionObserver((entries) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        document.body.classList.add('theme-light');
                    } else if (entry.boundingClientRect.top > 0) {
                        document.body.classList.remove('theme-light');
                    }
                });
            }, { rootMargin: '0px 0px -50% 0px' });
            const trigger = document.querySelector('.theme-trigger-trust');
            if (trigger) themeObserver.observe(trigger);

            // Footer Logic
            const siteFooter = document.querySelector('.site-footer');
            if (siteFooter) {
                const footerObserver = new IntersectionObserver((entries) => {
                    entries.forEach(entry => {
                        if (entry.isIntersecting) {
                            siteFooter.classList.add('revealed');
                        }
                    });
                }, { rootMargin: '0px 0px -10% 0px' });
                footerObserver.observe(siteFooter);
                
                const backBtn = document.getElementById('backToTop');
                if (backBtn) {
                    backBtn.addEventListener('click', (e) => {
                        e.preventDefault();
                        window.scrollTo({ top: 0, behavior: 'smooth' });
                    });
                }
            }

            const themeFooterObserver = new IntersectionObserver((entries) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        document.body.classList.remove('theme-light');
                    } else if (entry.boundingClientRect.top > 0) {
                        document.body.classList.add('theme-light');
                    }
                });
            }, { rootMargin: '0px 0px -50% 0px' });
            const triggerFooter = document.querySelector('.theme-trigger-footer');
            if (triggerFooter) themeFooterObserver.observe(triggerFooter);

        })();

            const TOTAL = EVENTS.length;
            const N_SLOTS = 7;    // positions 0–6, center = slot 3
            const CENTER = 3;
            const HOLD_MS = 5000; // 5 second hold
            const TRANS_MS = 1200; // transition duration

            /* ─── State ────────────────────────────────────────────── */
            let centerIdx = 0;   // which event is in the center slot
            let isAnim = false;
            let isPaused = false;
            let autoTimer = null;

            /* ─── DOM refs ─────────────────────────────────────────── */
            const stage = document.getElementById('cardsStage');
            const zone = document.getElementById('carouselZone');

            /* Circular array of card elements (length = N_SLOTS) */
            const cards = [];

            /* ─── Helpers ──────────────────────────────────────────── */
            function mod(n, m) { return ((n % m) + m) % m; }

            /* Get the event index for a given slot position */
            function eventForSlot(slot) {
                const offset = slot - CENTER;              // -3 … +3
                return mod(centerIdx + offset, TOTAL);
            }

            /* Fill a card element with event data */
            function fillCard(el, ev) {
                el.querySelector('img').src = ev.img;
                el.querySelector('img').alt = ev.alt;
                el.querySelector('.c-num').textContent = ev.num;
                el.querySelector('.c-name').textContent = ev.name;
                el.querySelector('.c-sub').textContent = ev.sub;
                el.setAttribute('aria-label', ev.name);
            }

            /* ─── Build all 7 cards ────────────────────────────────── */
            function buildCards() {
                for (let slot = 0; slot < N_SLOTS; slot++) {
                    const ev = EVENTS[eventForSlot(slot)];
                    const card = document.createElement('article');
                    card.className = 'event-card';
                    card.setAttribute('data-slot', slot);
                    card.setAttribute('aria-label', ev.name);

                    card.innerHTML = `
                <img src="${ev.img}" alt="${ev.alt}" loading="${slot === CENTER ? 'eager' : 'lazy'}">
                <div class="card-info">
                    <div class="c-num">${ev.num}</div>
                    <div class="c-name">${ev.name}</div>
                    <div class="c-sub">${ev.sub}</div>
                </div>
            `;

                    /* Click adjacent cards to navigate */
                    card.addEventListener('click', () => {
                        const s = parseInt(card.getAttribute('data-slot'), 10);
                        if (s === CENTER - 1) manualAdvance(-1);
                        else if (s === CENTER + 1) manualAdvance(+1);
                    });

                    stage.appendChild(card);
                    cards.push(card);
                }
            }

            /* ─── ADVANCE ──────────────────────────────────────────── */
            /*
             * direction: +1 = forward (Weddings → Corporate → …)
             *            -1 = backward
             *
             * Algorithm (forward):
             *   1. cards[0] is the leftmost (slot 0).
             *   2. Disable its transition, move to slot 7 (off-screen right) instantly.
             *   3. Update its content for the new rightmost event.
             *   4. Force reflow, re-enable transition.
             *   5. Shift the cards array: [0,1,2,3,4,5,6] → [1,2,3,4,5,6,0-recycled]
             *   6. Update all data-slot attributes → CSS transitions fire simultaneously.
             */
            function advance(dir) {
                if (isAnim) return;
                isAnim = true;

                /* Update center event index */
                centerIdx = mod(centerIdx + dir, TOTAL);

                if (dir > 0) {
                    /* Forward: recycle leftmost card to right */
                    const recycling = cards[0];

                    /* Teleport it off-screen right (no transition) */
                    recycling.style.transition = 'none';
                    recycling.setAttribute('data-slot', '7');

                    /* Update content for new rightmost slot */
                    fillCard(recycling, EVENTS[eventForSlot(N_SLOTS - 1)]);

                    /* Force browser reflow to register the teleport */
                    recycling.offsetHeight; // eslint-disable-line

                    /* Re-enable transitions */
                    recycling.style.transition = '';

                    /* Rotate the array: shift leftmost to rightmost */
                    cards.push(cards.shift());

                } else {
                    /* Backward: recycle rightmost card to left */
                    const recycling = cards[N_SLOTS - 1];

                    recycling.style.transition = 'none';
                    recycling.setAttribute('data-slot', '-1');

                    fillCard(recycling, EVENTS[eventForSlot(0)]);

                    recycling.offsetHeight; // eslint-disable-line

                    recycling.style.transition = '';

                    cards.unshift(cards.pop());
                }

                /* Re-assign all slot indices → all CSS transitions fire */
                cards.forEach((card, idx) => {
                    card.setAttribute('data-slot', idx);
                });

                setTimeout(() => { isAnim = false; }, TRANS_MS + 50);
            }

            function manualAdvance(dir) {
                pauseTemporarily(4000);
                advance(dir);
            }

            function pauseTemporarily(ms) {
                isPaused = true;
                clearTimeout(pauseTemporarily._t);
                pauseTemporarily._t = setTimeout(() => { isPaused = false; }, ms);
            }



            /* ─── Autoplay ─────────────────────────────────────────── */
            function startAuto() {
                clearInterval(autoTimer);
                autoTimer = setInterval(() => {
                    if (!isPaused) advance(+1);
                }, HOLD_MS);
            }

            /* ─── Hover pause ──────────────────────────────────────── */
            zone.addEventListener('mouseenter', () => { isPaused = true; });
            zone.addEventListener('mouseleave', () => { isPaused = false; });

            /* ─── Drag / Swipe ─────────────────────────────────────── */
            let dragX = 0;
            const THRESH = 55;

            zone.addEventListener('mousedown', e => { dragX = e.clientX; });
            zone.addEventListener('mouseup', e => {
                const d = dragX - e.clientX;
                if (Math.abs(d) > THRESH) manualAdvance(d > 0 ? +1 : -1);
            });
            zone.addEventListener('touchstart', e => { dragX = e.touches[0].clientX; }, { passive: true });
            zone.addEventListener('touchend', e => {
                const d = dragX - e.changedTouches[0].clientX;
                if (Math.abs(d) > THRESH) manualAdvance(d > 0 ? +1 : -1);
            }, { passive: true });

            /* ─── Keyboard ─────────────────────────────────────────── */
            document.addEventListener('keydown', e => {
                if (e.key === 'ArrowRight') manualAdvance(+1);
                if (e.key === 'ArrowLeft') manualAdvance(-1);
            });

            /* ─── Init ─────────────────────────────────────────────── */
            buildCards();

            window.addEventListener('load', () => {
                const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
                if (!reducedMotion) {
                    startAuto();
                    zone.addEventListener('mouseenter', () => { isPaused = true; });
                    zone.addEventListener('mouseleave', () => { isPaused = false; });
                }
            });

        })();

        (function () {
            'use strict';
            const panels = document.querySelectorAll('.d-panel');
            if (!panels.length) return;

            let activePanel = panels[0];

            function setActive(panel) {
                if (activePanel === panel) return;
                activePanel.classList.remove('active');
                panel.classList.add('active');
                activePanel = panel;

                // For mobile: ensure the active panel is scrolled into view smoothly
                if (window.innerWidth <= 767) {
                    panel.scrollIntoView({ behavior: 'smooth', inline: 'center', block: 'nearest' });
                }
            }

            panels.forEach(panel => {
                panel.addEventListener('mouseenter', () => {
                    if (window.innerWidth > 767) setActive(panel);
                });

                panel.addEventListener('focus', () => {
                    setActive(panel);
                });

                panel.addEventListener('click', () => {
                    setActive(panel);
                });

                // Keyboard accessibility
                panel.addEventListener('keydown', (e) => {
                    if (e.key === 'Enter' || e.key === ' ') {
                        e.preventDefault();
                        setActive(panel);
                    }
                });
            });

        })();

        // Theme observer
        const themeObserver = new IntersectionObserver((entries) => {
            entries.forEach(entry => {
                if (entry.isIntersecting) {
                    document.body.classList.add('theme-light');

                    // Auto snap the section to the top of the screen
                    const discoverySection = document.querySelector('.event-discovery');
                    if (discoverySection) {
                        setTimeout(() => {
                            discoverySection.scrollIntoView({ behavior: 'smooth', block: 'start' });
                        }, 50);
                    }
                } else {
                    // Only revert if we scroll back UP
                    if (entry.boundingClientRect.top > 0) {
                        document.body.classList.remove('theme-light');
                    }
                }
            });
        }, {
            root: null,
            rootMargin: '0px 0px -50% 0px', // Triggers when the trigger hits the middle of the screen
            threshold: 0
        });

        const trigger = document.querySelector('.theme-trigger');
        if (trigger) {
            themeObserver.observe(trigger);
